# 并行调用工具

某些语言模型支持并行调用工具。当多个工具彼此独立且可以在同一生成步骤中并行执行时，这尤其有用。

<Browser>
  <ChatGeneration
    history={[
      { role: 'User', content: '最近怎么样？' },
      { role: 'Assistant', content: '一切都好，有什么可以帮您的吗？' },
    ]}
    inputMessage={{
      role: 'User',
      content: '巴黎和纽约的天气怎么样？',
    }}
    outputMessage={{
      role: 'Assistant',
      content:
        '纽约气温 24°C，巴黎气温 25°C。两个城市都是晴天。',
    }}
  />
</Browser>

## 客户端

让我们修改之前的示例，并行调用 `getWeather` 工具来获取多个城市的天气。

```tsx filename='app/page.tsx'
'use client';

import { useState } from 'react';
import { Message, continueConversation } from './actions';

// 允许流式响应最长 30 秒
export const maxDuration = 30;

export default function Home() {
  const [conversation, setConversation] = useState<Message[]>([]);
  const [input, setInput] = useState<string>('');

  return (
    <div>
      <div>
        {conversation.map((message, index) => (
          <div key={index}>
            {message.role}: {message.content}
          </div>
        ))}
      </div>

      <div>
        <input
          type="text"
          value={input}
          onChange={event => {
            setInput(event.target.value);
          }}
        />
        <button
          onClick={async () => {
            const { messages } = await continueConversation([
              ...conversation,
              { role: 'user', content: input },
            ]);

            setConversation(messages);
          }}
        >
          发送消息
        </button>
      </div>
    </div>
  );
}
```

## 服务器

让我们更新工具对象，改为使用 `getWeather` 函数。

```ts filename='app/actions.ts'
'use server';

import { generateText } from 'ai';
import { z } from 'zod';

export interface Message {
  role: 'user' | 'assistant';
  content: string;
}

function getWeather({ city, unit }) {
  // 此函数通常会发出
  // API 请求来获取天气。

  return { value: 25, description: 'Sunny' };
}

export async function continueConversation(history: Message[]) {
  'use server';

  const { text, toolResults } = await generateText({
    model: 'openai/gpt-5.4',
    instructions: '你是一个友好的天气助手！',
    messages: history,
    tools: {
      getWeather: {
        description: '获取某个地点的天气',
        inputSchema: z.object({
          city: z.string().describe('要获取天气的城市'),
          unit: z
            .enum(['C', 'F'])
            .describe('显示温度的单位'),
        }),
        execute: async ({ city, unit }) => {
          const weather = getWeather({ city, unit });
          return `当前 ${city} 气温 ${weather.value}°${unit}，天气${weather.description}！`;
        },
      },
    },
  });

  return {
    messages: [
      ...history,
      {
        role: 'assistant' as const,
        content:
          text || toolResults.map(toolResult => toolResult.result).join('\n'),
      },
    ],
  };
}
```
