# 调用工具

某些模型允许开发者提供一组工具，这些工具可以在生成过程中的任何时间被调用。这对于扩展语言模型的能力非常有用，使其能够利用逻辑或数据与模型外部的系统进行交互。

<Browser>
  <ChatGeneration
    history={[
      { role: 'User', content: '最近怎么样？' },

    ]}
    inputMessage={{
      role: 'User',

## 客户端

让我们创建一个用户与模型之间的简单对话，并放置一个按钮来调用 `continueConversation`。

```tsx filename='app/page.tsx'
'use client';

import { useState } from 'react';
import { Message, continueConversation } from './actions';

// 允许流式响应最长 30 秒
export const maxDuration = 30;

export default function Home() {
  const [conversation, setConversation] = useState<Message[]>([]);
  const [input, setInput] = useState<string>('');

  return (
    <div>
      <div>
        {conversation.map((message, index) => (
          <div key={index}>
            {message.role}: {message.content}
          </div>
        ))}
      </div>

      <div>
        <input
          type="text"
          value={input}
          onChange={event => {
            setInput(event.target.value);
          }}
        />
        <button
          onClick={async () => {
            const { messages } = await continueConversation([
              ...conversation,
              { role: 'user', content: input },
            ]);

            setConversation(messages);
          }}
        >
          发送消息
        </button>
      </div>
    </div>
  );
}
```

## 服务器

现在，让我们实现 `continueConversation` 操作，使用 `generateText` 来生成对用户问题的响应。我们将使用 [`tools`](/docs/reference/ai-sdk-core/generate-text#tools) 参数来指定我们自己的函数 `celsiusToFahrenheit`，该函数将把用户给定的摄氏温度值转换为华氏温度。

我们将使用 zod 来指定 `celsiusToFahrenheit` 函数参数的架构。

```tsx filename='app/actions.ts'
'use server';

import { generateText } from 'ai';
import { z } from 'zod';

export interface Message {
  role: 'user' | 'assistant';
  content: string;
}

export async function continueConversation(history: Message[]) {
  'use server';

  const { text, toolResults } = await generateText({
    model: 'openai/gpt-5.4',
    instructions: '你是一位友好的助手！',
    messages: history,
    tools: {
      celsiusToFahrenheit: {
        description: '将摄氏温度转换为华氏温度',
        inputSchema: z.object({
          value: z.string().describe('摄氏温度值'),
        }),
        execute: async ({ value }) => {
          const celsius = parseFloat(value);
          const fahrenheit = celsius * (9 / 5) + 32;
          return `${celsius}°C 是 ${fahrenheit.toFixed(2)}°F`;
        },
      },
    },
  });

  return {
    messages: [
      ...history,
      {
        role: 'assistant' as const,
        content:
          text || toolResults.map(toolResult => toolResult.result).join('\n'),
      },
    ],
  };
}
```
