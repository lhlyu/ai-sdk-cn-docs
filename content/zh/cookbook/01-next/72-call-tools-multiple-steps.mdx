# 多步骤工具调用

某些语言模型擅长通过多步骤工具调用来完成更复杂的任务。当工具之间存在依赖关系，需要在同一生成步骤中按顺序执行时，这一特性尤为实用。

## 客户端

让我们创建一个 React 组件，从 `@ai-sdk/react` 模块导入 `useChat` Hook。当用户发送消息时，`useChat` Hook 会调用 `/api/chat` 端点。该端点将基于对话历史生成助手的响应，并将其流式传输到客户端。如果助手响应中包含工具调用，Hook 也会自动显示这些调用。

```tsx filename='app/page.tsx'
'use client';

import { useChat } from '@ai-sdk/react';
import { DefaultChatTransport } from 'ai';
import { useState } from 'react';
import type { ChatMessage } from './api/chat/route';

export default function Page() {
  const [input, setInput] = useState('');

  const { messages, sendMessage } = useChat<ChatMessage>({
    transport: new DefaultChatTransport({
      api: '/api/chat',
    }),
  });

  return (
    <div>
      <input
        className="border"
        value={input}
        onChange={event => {
          setInput(event.target.value);
        }}
        onKeyDown={async event => {
          if (event.key === 'Enter') {
            sendMessage({
              text: input,
            });
            setInput('');
          }
        }}
      />

      {messages.map((message, index) => (
        <div key={index}>
          {message.parts.map((part, i) => {
            switch (part.type) {
              case 'text':
                return <div key={`${message.id}-text`}>{part.text}</div>;
              case 'tool-getLocation':
              case 'tool-getWeather':
                return (
                  <div key={`${message.id}-weather-${i}`}>
                    {JSON.stringify(part, null, 2)}
                  </div>
                );
            }
          })}
        </div>
      ))}
    </div>
  );
}
```

## 服务端

你需要在 `/api/chat` 创建一个新路由，使用 `ai` 模块中的 `streamText` 函数，基于对话历史生成助手的响应。

你将使用 [`tools`](/docs/reference/ai-sdk-core/generate-text#tools) 参数指定两个工具：`getLocation` 和 `getWeather`。这两个工具会先获取用户位置，然后利用该位置获取天气信息。

你需要添加前面提到的两个函数，并使用 zod 指定其参数的模式。

要实现多步骤工具调用，可以使用 `stopWhen` 选项来指定模型生成工具调用时的停止条件。在此示例中，你将设置为 `isStepCount(5)`，以允许多次连续的工具调用（步骤）。

```ts filename='app/api/chat/route.ts'
import {
  type InferUITools,
  type ToolSet,
  type UIDataTypes,
  type UIMessage,
  convertToModelMessages,
  createUIMessageStreamResponse,
  isStepCount,
  streamText,
  toUIMessageStream,
  tool,
} from 'ai';
import { z } from 'zod';

const tools = {
  getLocation: tool({
    description: '获取用户的位置',
    inputSchema: z.object({}),
    execute: async () => {
      const location = { lat: 37.7749, lon: -122.4194 };
      return `你的位置位于纬度 ${location.lat}，经度 ${location.lon}`;
    },
  }),
  getWeather: tool({
    description: '获取某个位置的天气',
    inputSchema: z.object({
      city: z.string().describe('要获取天气的城市'),
      unit: z
        .enum(['C', 'F'])
        .describe('显示温度的单位'),
    }),
    execute: async ({ city, unit }) => {
      const weather = {
        value: 24,
        description: '晴朗',
      };

      return `当前 ${city} 的天气为 ${weather.value}°${unit}，${weather.description}！`;
    },
  }),
} satisfies ToolSet;

export type ChatTools = InferUITools<typeof tools>;

export type ChatMessage = UIMessage<never, UIDataTypes, ChatTools>;

export async function POST(req: Request) {
  const { messages }: { messages: ChatMessage[] } = await req.json();

  const result = streamText({
    model: 'openai/gpt-4o',
    system: '你是一个乐于助人的助手。',
    messages: await convertToModelMessages(messages),
    stopWhen: isStepCount(5),
    tools,
  });

  return createUIMessageStreamResponse({
    stream: toUIMessageStream({ stream: result.stream }),
  });
}
```
